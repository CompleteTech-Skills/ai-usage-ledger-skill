+// Nonblocking/no-follow open prevents a replaced config from becoming a FIFO
+// wait or symlink read between path validation and opening.
+func openConfig(path string) (*os.File, error) {
+	before, err := os.Lstat(path)
+	if err != nil {
+		return nil, err
+	}
+	f, err := os.OpenFile(path, os.O_RDONLY|syscall.O_NOFOLLOW|syscall.O_NONBLOCK, 0)
+	if err != nil {
+		return nil, err
+	}
+	after, err := f.Stat()
+	if err != nil || !after.Mode().IsRegular() || after.Mode().Perm()&0077 != 0 || !os.SameFile(before, after) {
+		_ = f.Close()
+		return nil, errors.New("config changed or is not a private regular file")
+	}
+	return f, nil
+}

+func privatePath(path string, dir bool) error {
+	if !filepath.IsAbs(path) {
+		return errors.New("export path must be absolute")
+	}
+	for p := filepath.Clean(path); ; p = filepath.Dir(p) {
+		st, err := os.Lstat(p)
+		if err != nil || st.Mode()&os.ModeSymlink != 0 {
+			return errors.New("unsafe export path")
+		}
+		if p == filepath.Clean(path) && (st.IsDir() != dir || !dir && !st.Mode().IsRegular() || st.Mode().Perm()&0077 != 0) {
+			return errors.New("export path must be private")
+		}
+		if filepath.Dir(p) == p {
+			break
+		}
+	}
+	return nil
+}

+func Open(configPath string) (*Exporter, error) {
+	if err := privatePath(configPath, false); err != nil {
+		return nil, err
+	}
+	f, err := openConfig(configPath)
+	if err != nil {
+		return nil, errors.New("cannot open export config")
+	}
+	defer f.Close()
+	dec := json.NewDecoder(io.LimitReader(f, 65537))
+	dec.DisallowUnknownFields()
+	var cfg Config
+	if dec.Decode(&cfg) != nil || !cfg.Approved || !identifier.MatchString(cfg.Source) || len(cfg.Projects) == 0 {
+		return nil, errors.New("invalid export config")
+	}
+	var extra any
+	if dec.Decode(&extra) != io.EOF {
+		return nil, errors.New("invalid export config trailing data")
+	}
+	if err := privatePath(cfg.Directory, true); err != nil {
+		return nil, err
+	}
+	projects := map[string]bool{}
+	for _, p := range cfg.Projects {
+		if !identifier.MatchString(p) || projects[p] {
+			return nil, errors.New("invalid project scope")
+		}
+		projects[p] = true
+	}
+	id, err := randomID()
+	if err != nil {
+		return nil, err
+	}
+	journal, err := os.OpenFile(filepath.Join(cfg.Directory, "journal-"+id+".jsonl"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
+	if err != nil {
+		return nil, errors.New("cannot create private export journal")
+	}
+	return &Exporter{file: journal, source: cfg.Source, projects: projects}, nil
+}
+func randomID() (string, error) {
+	var b [16]byte
+	_, err := rand.Read(b[:])
+	return hex.EncodeToString(b[:]), err

+func (e *Exporter) Close() error { e.mu.Lock(); defer e.mu.Unlock(); return e.file.Close() }
+func (e *Exporter) write(row map[string]any) error {
+	b, err := json.Marshal(row)
+	if err != nil {
+		return err
+	}
+	e.mu.Lock()
+	defer e.mu.Unlock()
+	if _, err = e.file.Write(append(b, '\n')); err != nil {
+		return err
+	}
+	return e.file.Sync()

+// Do wraps precisely one http.Client.Do invocation after provider selection.
+// Export failures never retry, cancel, or replace the model request.
+func Do(ctx context.Context, client *http.Client, req *http.Request, provider, model string) (*http.Response, error) {
+	e := selected(ctx)
+	scope, _ := ctx.Value(scopeKey{}).(Scope)
+	if e == nil || !e.projects[scope.Project] || !identifier.MatchString(scope.Plan) || !identifier.MatchString(provider) || !modelName.MatchString(model) {
+		return client.Do(req)
+	}
+	for _, s := range []string{scope.Session, scope.Execution, scope.Step, scope.Build} {
+		if s != "" && !identifier.MatchString(s) {
+			return client.Do(req)
+		}
+	}
+	id, err := randomID()
+	if err != nil {
+		return client.Do(req)
+	}
+	scopeBytes, _ := json.Marshal(scope)
+	row := map[string]any{}
+	_ = json.Unmarshal(scopeBytes, &row)
+	row["schema"], row["kind"], row["granularity"] = "plandex-usage/1", "attempt_start", "attempt"
+	row["source_id"], row["plandex_version"], row["attempt_id"] = e.source, Version, id
+	row["provider"], row["model"], row["ts"] = provider, model, time.Now().UTC().Format(time.RFC3339Nano)
+	row["usage_basis"], row["token_semantics"], row["usage"], row["native_cost"] = "unavailable", "inclusive_input_output", nil, nil
+	if e.write(row) != nil {
+		log.Print("usage export unavailable: journal write failed")
+		return client.Do(req)
+	}
+	body := &observedBody{ctx: ctx, exporter: e, row: row}
+	resp, err := client.Do(req)
+	if err != nil {
+		body.finish("error")
+		return resp, err
+	}
+	if resp.StatusCode < 200 || resp.StatusCode >= 400 {
+		body.finish("error")
+		return resp, nil
+	}
+	endpoint := req.URL
+	if resp.Request != nil {
+		endpoint = resp.Request.URL
+	}
+	body.direct = directEndpoint(provider, endpoint)
+	body.body = resp.Body
+	resp.Body = body
+	return resp, nil
+}
+func directEndpoint(provider string, endpoint *url.URL) bool {
+	return endpoint != nil && endpoint.Scheme == "https" && ((provider == "openai" && endpoint.Host == "api.openai.com") || (provider == "openrouter" && endpoint.Host == "openrouter.ai"))
+}

+type observedBody struct {
+	ctx                                                 context.Context
+	exporter                                            *Exporter
+	body                                                io.ReadCloser
+	mu                                                  sync.Mutex
+	row                                                 map[string]any
+	line                                                []byte
+	dropping, finished, direct, done, stopped, badUsage bool
+}
+
+func (b *observedBody) Read(p []byte) (int, error) {
+	n, err := b.body.Read(p)
+	b.mu.Lock()
+	for _, c := range p[:n] {
+		if c == '\n' {
+			if !b.dropping {
+				b.observe(bytes.TrimSpace(b.line))
+			}
+			b.line = b.line[:0]
+			b.dropping = false
+		} else if len(b.line) >= 65536 {
+			b.dropping = true
+			b.line = b.line[:0]
+			b.badUsage = true
+			b.row["usage"], b.row["usage_basis"] = nil, "unavailable"
+		} else if !b.dropping {
+			b.line = append(b.line, c)
+		}
+	}
+	b.mu.Unlock()
+	if err != nil {
+		status := "error"
+		if err == io.EOF {
+			status = "interrupted"
+		}
+		b.finish(status)
+	}
+	return n, err
+}
+func (b *observedBody) Close() error { err := b.body.Close(); b.finish("interrupted"); return err }
+func (b *observedBody) observe(line []byte) {
+	if b.finished || b.done || !bytes.HasPrefix(line, []byte("data:")) {
+		return
+	}
+	data := bytes.TrimSpace(bytes.TrimPrefix(line, []byte("data:")))
+	if bytes.Equal(data, []byte("[DONE]")) {
+		b.done = true
+		return
+	}
+	var chunk struct {
+		ID      string          `json:"id"`
+		Model   string          `json:"model"`
+		Usage   json.RawMessage `json:"usage"`
+		Error   json.RawMessage `json:"error"`
+		Choices []struct {
+			Finish string `json:"finish_reason"`
+		} `json:"choices"`
+	}
+	if json.Unmarshal(data, &chunk) != nil {
+		return
+	}
+	if len(chunk.Error) > 0 && string(chunk.Error) != "null" {
+		b.row["stream_error"] = true
+	}
+	for _, choice := range chunk.Choices {
+		if choice.Finish != "" {
+			b.stopped = true
+			if choice.Finish == "error" {
+				b.row["stream_error"] = true
+			}
+		}
+	}
+	if identifier.MatchString(chunk.ID) {
+		b.row["native_request_id"] = chunk.ID
+	}
+	if modelName.MatchString(chunk.Model) {
+		b.row["model"] = chunk.Model
+	}
+	if len(chunk.Usage) == 0 || string(chunk.Usage) == "null" {
+		return
+	}
+	usage, raw, ok := parseUsage(chunk.Usage)
+	if !ok {
+		b.badUsage = true
+		b.row["usage"], b.row["usage_basis"] = nil, "unavailable"
+		return
+	}
+	if b.badUsage {
+		return
+	}
+	b.row["usage"], b.row["raw_usage"] = usage, raw
+	var costFields map[string]json.RawMessage
+	_ = json.Unmarshal(chunk.Usage, &costFields)
+	if cost, ok := costFields["cost"]; ok {
+		var amount float64
+		if json.Unmarshal(cost, &amount) == nil && string(cost) != "null" && !math.IsInf(amount, 0) && amount >= 0 && amount <= 1e12 {
+			b.row["native_cost"] = map[string]string{"amount": string(cost), "currency": "USD", "provenance": "provider_response"}
+		}
+	}
+	b.row["usage_basis"] = "gateway_reported_unverified"
+	if b.direct {
+		b.row["usage_basis"] = "provider_reported"
+	}
+}
+func (b *observedBody) finish(status string) {
+	b.mu.Lock()
+	defer b.mu.Unlock()
+	if b.finished {
+		return
+	}
+	b.finished = true
+	if b.ctx.Err() != nil {
+		status = "cancelled"
+	} else if b.row["stream_error"] == true {
+		status = "error"
+	} else if b.done || b.stopped {
+		status = "completed"
+	}
+	delete(b.row, "stream_error")
+	b.row["kind"], b.row["status"], b.row["finished_at"] = "attempt", status, time.Now().UTC().Format(time.RFC3339Nano)
+	if b.exporter.write(b.row) != nil {
+		log.Print("usage export incomplete: journal write failed")
+	}
+	// Discard transient response-line storage on completion.
+	b.line = nil
+}

+func parseUsage(data []byte) (map[string]any, map[string]any, bool) {
+	var raw map[string]json.RawMessage
+	if json.Unmarshal(data, &raw) != nil || raw == nil {
+		return nil, nil, false
+	}
+	read := func(m map[string]json.RawMessage, key string) (*int64, bool) {
+		v, ok := m[key]
+		if !ok || string(v) == "null" {
+			return nil, true
+		}
+		var n int64
+		if json.Unmarshal(v, &n) != nil || n < 0 || n > maxCount {
+			return nil, false
+		}
+		return &n, true
+	}
+	fields := map[string]any{}
+	evidence := map[string]any{}
+	for target, name := range map[string]string{"input_tokens": "prompt_tokens", "output_tokens": "completion_tokens"} {
+		n, ok := read(raw, name)
+		if !ok {
+			return nil, nil, false
+		}
+		fields[target] = n
+		evidence[name] = n
+	}
+	for details, names := range map[string]map[string]string{"prompt_tokens_details": {"cache_read_tokens": "cached_tokens", "cache_write_tokens": "cache_creation_tokens"}, "completion_tokens_details": {"reasoning_tokens": "reasoning_tokens"}} {
+		var nested map[string]json.RawMessage
+		if v, ok := raw[details]; ok && json.Unmarshal(v, &nested) != nil {
+			return nil, nil, false
+		}
+		for target, name := range names {
+			n, ok := read(nested, name)
+			if !ok {
+				return nil, nil, false
+			}
+			fields[target] = n
+			evidence[map[string]string{"cache_read_tokens": "cached_tokens", "cache_write_tokens": "cache_write_tokens", "reasoning_tokens": "reasoning_tokens"}[target]] = n
+		}
+		if details == "prompt_tokens_details" {
+			// OpenRouter's documented cache-write field; other adapters may use
+			// cache_creation_tokens. Treat them as aliases, never extra tokens.
+			if _, exists := nested["cache_write_tokens"]; exists {
+				n, ok := read(nested, "cache_write_tokens")
+				old := fields["cache_write_tokens"].(*int64)
+				if !ok || old != nil && n != nil && *old != *n {
+					return nil, nil, false
+				}
+				if n != nil {
+					fields["cache_write_tokens"] = n
+					evidence["cache_write_tokens"] = n
+				}
+			}
+		}
+	}
+	for target, alias := range map[string]string{"cache_read_tokens": "cache_read_input_tokens", "cache_write_tokens": "cache_creation_input_tokens"} {
+		if _, exists := raw[alias]; exists {
+			n, ok := read(raw, alias)
+			if !ok {
+				return nil, nil, false
+			}
+			old := fields[target].(*int64)
+			if old != nil && n != nil && *old != *n {
+				return nil, nil, false
+			}
+			if n != nil {
+				fields[target] = n
+			}
+			evidence[alias] = n
+		}
+	}
+	in, out := fields["input_tokens"].(*int64), fields["output_tokens"].(*int64)
+	cr, cw, r := fields["cache_read_tokens"].(*int64), fields["cache_write_tokens"].(*int64), fields["reasoning_tokens"].(*int64)
+	value := func(n *int64) int64 {
+		if n == nil {
+			return 0
+		}
+		return *n
+	}
+	if in != nil && value(cr)+value(cw) > *in || out != nil && value(r) > *out {
+		return nil, nil, false
+	}
+	if in != nil && out != nil && *in+*out > maxCount {
+		return nil, nil, false
+	}
+	if total, ok := read(raw, "total_tokens"); !ok || total != nil && in != nil && out != nil && *total != *in+*out {
+		return nil, nil, false
+	} else {
+		evidence["total_tokens"] = total
+	}
+	return fields, evidence, true
+}

+func rows(t *testing.T, dir string) []map[string]any {
+	t.Helper()
+	files, _ := filepath.Glob(filepath.Join(dir, "journal-*.jsonl"))
+	var result []map[string]any
+	for _, p := range files {
+		b, err := os.ReadFile(p)
+		if err != nil {
+			t.Fatal(err)
+		}
+		for _, line := range strings.Split(strings.TrimSpace(string(b)), "\n") {
+			if line == "" {
+				continue
+			}
+			var r map[string]any
+			if json.Unmarshal([]byte(line), &r) != nil {
+				t.Fatal("invalid journal")
+			}
+			result = append(result, r)
+		}
+	}
+	return result
